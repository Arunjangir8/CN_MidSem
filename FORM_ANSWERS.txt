# ===== Team team1 edge (Mac 2) - PHASE 1 =====
# Installed as: /opt/homebrew/etc/nginx/servers/team.conf
# (Homebrew's nginx.conf already does include servers/*; inside http{})

log_format team_lb '$remote_addr:$remote_port "$request" $status '
                   'upstream=$upstream_addr up_status=$upstream_status '
                   'tls=$ssl_protocol rt=$request_time';

# The pool of backends. Default algorithm = round robin.
# (alternative: add least_conn; and explain it in the viva)
upstream team_backends {
    server 10.7.5.55:3001;   # Backend A (Mac 3)
    server 10.7.16.171:3002;   # Backend B (Mac 4)
}

# Plain HTTP -> redirect to HTTPS
server {
    listen 80;
    server_name app.team1.test api.team1.test;
    return 301 https://$host:443$request_uri;
}

# HTTPS: TLS terminates HERE; backends receive plain HTTP on the LAN
server {
    listen 443 ssl;
    http2 on;                               # nginx >= 1.25.1
    server_name app.team1.test api.team1.test;

    ssl_certificate     /opt/homebrew/etc/nginx/certs/server.crt;
    ssl_certificate_key /opt/homebrew/etc/nginx/certs/server.key;
    ssl_protocols       TLSv1.2 TLSv1.3;

    access_log /tmp/nginx-team-access.log team_lb;
    error_log  /tmp/nginx-team-error.log  warn;

    add_header X-Edge     $hostname       always;  # which edge answered
    add_header X-Upstream $upstream_addr  always;  # which backend IP:port

    location / {
        proxy_pass http://team_backends;
        proxy_http_version 1.1;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Connection        "";
    }
}

==================== D2 ====================
Cache-Control: public, max-age=60 on /api/catalog tells the client (and any shared cache) that the response may be stored and reused for 60 seconds counted from the Date header, so for the next 60 s the browser serves /api/catalog from its own cache without contacting our edge at all (DevTools shows "(disk cache)"). After the 60 s TTL expires the copy becomes stale, so the client revalidates by sending a conditional request with If-None-Match: "55df56e6b3ea9005" (the stored ETag). If the resource has not changed, the server answers 304 Not Modified with headers only and no body, and the client keeps using its cached copy for another 60 s; if the ETag no longer matches, the server returns a fresh 200 with the new body and new ETag. We verified this: the conditional request above returned HTTP/2 304 with the same ETag, while a wrong ETag returns 200. Both backends compute the same ETag from the content, so the 304 works no matter which backend the load balancer picks. In contrast, /api/status uses no-store, so it is never cached and always shows live load-balancer data.
